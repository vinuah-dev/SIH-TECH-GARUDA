# --- ANPR ----------------------------------------------------------------
# Optional. Without either of these IBVAP still locates plates and saves the
# crop as evidence; it simply cannot read the number. First use downloads
# model weights.
#
# PaddleOCR is the default and was measured far better on real Indian plates -
# 73.2% of plates read exactly against EasyOCR's 34.5%, and faster with it
# (0.22s a plate against 0.56s), because only its recogniser is used and the
# plate is already located. See app/anpr/readers.py.

paddleocr>=3.0

# Fallback recogniser, used automatically when PaddleOCR will not load.
easyocr>=1.7
